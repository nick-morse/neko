432 subroutine bc_finalize_base(this, only_facets)
433 class(bc_t), target, intent(inout) :: this
434 logical, optional, intent(in) :: only_facets
435 type(tuple_i4_t), pointer :: bfp(:)
436 type(tuple_i4_t) :: bc_facet
437 type(field_t) :: test_field
438 integer :: facet_size, facet, el
439 logical :: only_facet = .false.
440 integer :: i, j, k, l, msk_c
441 integer :: lx, ly, lz, n
442 lx = this%Xh%lx
443 ly = this%Xh%ly
444 lz = this%Xh%lz
445 if ( present(only_facets)) then
446 only_facet = only_facets
447 else
448 only_facet = .false.
449 end if
451
452 ! Note we assume that lx = ly = lz
453 facet_size = lx**2
454 allocate(this%msk(0:facet_size * this%marked_facet%size()))
455 allocate(this%facet(0:facet_size * this%marked_facet%size()))
456
457 msk_c = 0
458 bfp => this%marked_facet%array()
459
460 ! Loop through each (facet, element) id tuple
461 ! Then loop over all the nodes of the face and compute their linear index
462 ! This index goes into this%msk, whereas the corresponding face id goes into
463 ! this%facet
464 do i = 1, this%marked_facet%size()
465 bc_facet = bfp(i)
466 facet = bc_facet%x(1)
467 el = bc_facet%x(2)
468 select case (facet)
469 case (1)
470 do l = 1, lz
471 do k = 1, ly
472 msk_c = msk_c + 1
473 this%msk(msk_c) = linear_index(1, k, l, el, lx, ly, lz)
474 this%facet(msk_c) = 1
475 end do
476 end do
477 case (2)
478 do l = 1, lz
479 do k = 1, ly
480 msk_c = msk_c + 1
481 this%msk(msk_c) = linear_index(lx, k, l, el, lx, ly, lz)
482 this%facet(msk_c) = 2
483 end do
484 end do
485 case (3)
486 do l = 1, lz
487 do j = 1, lx
488 msk_c = msk_c + 1
489 this%msk(msk_c) = linear_index(j, 1, l, el, lx, ly, lz)
490 this%facet(msk_c) = 3
491 end do
492 end do
493 case (4)
494 do l = 1, lz
495 do j = 1, lx
496 msk_c = msk_c + 1
497 this%msk(msk_c) = linear_index(j, ly, l, el, lx, ly, lz)
498 this%facet(msk_c) = 4
499 end do
500 end do
501 case (5)
502 do k = 1, ly
503 do j = 1, lx
504 msk_c = msk_c + 1
505 this%msk(msk_c) = linear_index(j, k, 1, el, lx, ly, lz)
506 this%facet(msk_c) = 5
507 end do
508 end do
509 case (6)
510 do k = 1, ly
511 do j = 1, lx
512 msk_c = msk_c + 1
513 this%msk(msk_c) = linear_index(j, k, lz, el, lx, ly, lz)
514 this%facet(msk_c) = 6
515 end do
516 end do
517 end select
518 end do
519 this%facet(0) = msk_c
520 if (neko_bcknd_device .eq. 1) then
521 !Observe the facet_mask is junk if only_facet is false
522 n = msk_c + 1
523 call device_map(this%facet, this%facet_d, n)
524 call device_memcpy(this%facet, this%facet_d, n, &
525 host_to_device, sync = .true.)
526 end if
527 if ( .not. only_facet) then
528 !Makes check for points not on facet that should have bc applied
529 call test_field%init(this%dof)
530
531 n = test_field%size()
532 test_field%x = 0.0_rp
533 !Apply this bc once
534 do i = 1, msk_c
535 test_field%x(this%msk(i),1,1,1) = 1.0
536 end do
537 if (neko_bcknd_device .eq. 1) then
538 call device_memcpy(test_field%x, test_field%x_d, n, &
539 host_to_device, sync=.true.)
540 end if
541 !Check if some point that was not zeroed was zeroed on another element
542 call this%coef%gs_h%op(test_field,gs_op_add)
543 if (neko_bcknd_device .eq. 1) then
544 call device_memcpy(test_field%x, test_field%x_d, n, &
545 device_to_host, sync=.true.)
546 end if
547 msk_c = 0
548 do i = 1, this%dof%size()
549 if (test_field%x(i,1,1,1) .gt. 0.5) then
550 msk_c = msk_c + 1
551 end if
552 end do
553 !Allocate new mask
554 deallocate(this%msk)
555 allocate(this%msk(0:msk_c))
556 j = 1
557 do i = 1, this%dof%size()
558 if (test_field%x(i,1,1,1) .gt. 0.5) then
559 this%msk(j) = i
560 j = j + 1
561 end if
562 end do
563
564 call test_field%free()
565 end if
566
567 this%msk(0) = msk_c
568
569 if (neko_bcknd_device .eq. 1) then
570 n = msk_c + 1
571 call device_map(this%msk, this%msk_d, n)
572 call device_memcpy(this%msk, this%msk_d, n, &
573 host_to_device, sync = .true.)
574 end if
575
576 end subroutine bc_finalize_base
