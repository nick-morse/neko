555 /*
556 * Reduction buffer
557 */
558 int red_s = 0;
559 real * bufred = NULL;
560 real * bufred_d = NULL;
561
562 void hip_redbuf_check_alloc(int nb) {
563 if ( nb >= red_s){
564 red_s = nb+1;
565 if (bufred != NULL) {
566 HIP_CHECK(hipHostFree(bufred));
567 HIP_CHECK(hipFree(bufred_d));
568 }
569 HIP_CHECK(hipHostMalloc(&bufred,red_s*sizeof(real),hipHostMallocDefault));
570 HIP_CHECK(hipMalloc(&bufred_d, red_s*sizeof(real)));
571 }
572 }

577 void hip_global_reduce_add(real * bufred, void * bufred_d, int n, const hipStream_t stream) {
578 #ifdef HAVE_RCCL
579 device_nccl_allreduce(bufred_d, bufred_d, n, sizeof(real),
580 DEVICE_NCCL_SUM, stream);
581 HIP_CHECK(hipMemcpyAsync(bufred, bufred_d, n*sizeof(real),
582 hipMemcpyDeviceToHost, stream));
583 HIP_CHECK(hipStreamSynchronize(stream));
584 #elif HAVE_DEVICE_MPI
585 HIP_CHECK(hipStreamSynchronize(stream));
586 device_mpi_allreduce(bufred_d, bufred, n, sizeof(real), DEVICE_MPI_SUM);
587 #else
588 HIP_CHECK(hipMemcpyAsync(bufred, bufred_d, n*sizeof(real),
589 hipMemcpyDeviceToHost, stream));
590 HIP_CHECK(hipStreamSynchronize(stream));
591 #endif
592 }

598 real hip_vlsc3(void *u, void *v, void *w, int *n) {
599
600 const dim3 nthrds(1024, 1, 1);
601 const dim3 nblcks(((*n)+1024 - 1)/ 1024, 1, 1);
602 const int nb = ((*n) + 1024 - 1)/ 1024;
603 const hipStream_t stream = (hipStream_t) glb_cmd_queue;
604 hip_redbuf_check_alloc(nb);
605
606
607 hipLaunchKernelGGL(HIP_KERNEL_NAME(glsc3_kernel<real>),
608 nblcks, nthrds, 0, stream,
609 (real *) u, (real *) v,
610 (real *) w, bufred_d, *n);
611 HIP_CHECK(hipGetLastError());
612 hipLaunchKernelGGL(HIP_KERNEL_NAME(reduce_kernel<real>),
613 1, 1024, 0, stream, bufred_d, nb);
614 HIP_CHECK(hipGetLastError());
615
616 HIP_CHECK(hipMemcpyAsync(bufred, bufred_d, sizeof(real),
617 hipMemcpyDeviceToHost, stream));
618 HIP_CHECK(hipStreamSynchronize(stream));
619
620 return bufred[0];
621 }

654 void hip_glsc3_many(real *h, void * w, void *v,void *mult, int *j, int *n){
655 int pow2 = 1;
656 while(pow2 < (*j)){
657 pow2 = 2*pow2;
658 }
659 const int nt = 1024/pow2;
660 const dim3 nthrds(pow2, nt, 1);
661 const dim3 nblcks(((*n)+nt - 1)/nt, 1, 1);
662 const dim3 nthrds_red(1024,1,1);
663 const dim3 nblcks_red( (*j),1,1);
664 const int nb = ((*n) + nt - 1)/nt;
665 const hipStream_t stream = (hipStream_t) glb_cmd_queue;
666 hip_redbuf_check_alloc((*j)*nb);
667 if (*n > 0) {
668 hipLaunchKernelGGL(HIP_KERNEL_NAME(glsc3_many_kernel<real>),
669 nblcks, nthrds, 0, stream,
670 (const real *) w, (const real **) v,
671 (const real *)mult, bufred_d, *j, *n);
672 HIP_CHECK(hipGetLastError());
673
674 hipLaunchKernelGGL(HIP_KERNEL_NAME(glsc3_reduce_kernel<real>),
675 nblcks_red, nthrds_red, 0, stream,
676 bufred_d, nb, *j);
677 HIP_CHECK(hipGetLastError());
678 } else { hip_rzero(bufred_d,&red_s); }
679 hip_global_reduce_add(h, bufred_d, (*j), stream);
680 }
