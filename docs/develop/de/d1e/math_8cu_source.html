541 /*
542 * Reduction buffer
543 */
544 int red_s = 0;
545 real * bufred = NULL;
546 void * bufred_d = NULL;
547
551 void cuda_redbuf_check_alloc(int nb) {
552 if ( nb >= red_s) {
553 red_s = nb+1;
554 if (bufred != NULL) {
555 CUDA_CHECK(cudaFreeHost(bufred));
556#ifdef HAVE_NVSHMEM
557 nvshmem_free(bufred_d);
558#else
559 CUDA_CHECK(cudaFree(bufred_d));
560#endif
561 }
562 CUDA_CHECK(cudaMallocHost(&bufred,red_s*sizeof(real)));
563#ifdef HAVE_NVSHMEM
564 bufred_d = (real *) nvshmem_malloc(red_s*sizeof(real));
565#else
566 CUDA_CHECK(cudaMalloc(&bufred_d, red_s*sizeof(real)));
567#endif
568 }
569 }

574 void cuda_global_reduce_add(real * bufred, void * bufred_d, int n, const cudaStream_t stream) {
575
576
577#ifdef HAVE_NCCL
578 device_nccl_allreduce(bufred_d, bufred_d, n, sizeof(real),
579 DEVICE_NCCL_SUM, stream);
580 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d, sizeof(real)*n,
581 cudaMemcpyDeviceToHost, stream));
582 cudaStreamSynchronize(stream);
583#elif HAVE_NVSHMEM
584 if (sizeof(real) == sizeof(float)) {
585 nvshmemx_float_sum_reduce_on_stream(NVSHMEM_TEAM_WORLD,
586 (float *) bufred_d,
587 (float *) bufred_d, n, stream);
588 }
589 else if (sizeof(real) == sizeof(double)) {
590 nvshmemx_double_sum_reduce_on_stream(NVSHMEM_TEAM_WORLD,
591 (double *) bufred_d,
592 (double *) bufred_d, n, stream);
593
594 }
595 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d,
596 sizeof(real)*n, cudaMemcpyDeviceToHost, stream));
597 cudaStreamSynchronize(stream);
598#elif HAVE_DEVICE_MPI
599 cudaStreamSynchronize(stream);
600 device_mpi_allreduce(bufred_d, bufred, n, sizeof(real), DEVICE_MPI_SUM);
601#else
602 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d, n*sizeof(real),
603 cudaMemcpyDeviceToHost, stream));
604 cudaStreamSynchronize(stream);
605#endif
606 }

614 real cuda_vlsc3(void *u, void *v, void *w, int *n) {
615
616 const dim3 nthrds(1024, 1, 1);
617 const dim3 nblcks(((*n)+1024 - 1)/ 1024, 1, 1);
618 const int nb = ((*n) + 1024 - 1)/ 1024;
619 const cudaStream_t stream = (cudaStream_t) glb_cmd_queue;
620
621 cuda_redbuf_check_alloc(nb);
622
623 glsc3_kernel<real><<<nblcks, nthrds, 0, stream>>>
624 ((real *) u, (real *) v, (real *) w, (real *) bufred_d, *n);
625 CUDA_CHECK(cudaGetLastError());
626 reduce_kernel<real><<<1, 1024, 0, stream>>> ((real *) bufred_d, nb);
627 CUDA_CHECK(cudaGetLastError());
628
629 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d, sizeof(real),
630 cudaMemcpyDeviceToHost, stream));
631 cudaStreamSynchronize(stream);
632
633 return bufred[0];
634 }

644

660
661 return bufred[0];
662 }

671 pow2 = 2*pow2;

674 const dim3 nthrds(pow2, nt, 1);

678 cuda_redbuf_check_alloc((*j)*nb);
679
680 if ( *n > 0) {
681 glsc3_many_kernel<real><<<nblcks, nthrds, 0, stream>>>
682 ((const real *) w, (const real **) v,
683 (const real *)mult, (real *)bufred_d, *j, *n);
684 CUDA_CHECK(cudaGetLastError());
685 glsc3_reduce_kernel<real>
686 <<<(*j), 1024, 0, stream>>>((real *) bufred_d, nb, *j);
687 CUDA_CHECK(cudaGetLastError());
688 } else { cuda_rzero(bufred_d,&red_s); }
689 cuda_global_reduce_add(h, bufred_d, (*j), stream);
690 }
