105 logical :: strong_ = .true.
106
107 if (present(strong)) strong_ = strong
108
109 if (.not. strong_) then
110 ! Compute the wall stress using the wall model.
111 call this%wall_model%compute(t, tstep)
112
113 ! Populate the 3D wall stress field for post-processing.

120 end if
121
122 ! Either add the stress to the RHS or apply the non-penetration condition
