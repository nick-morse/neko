145 subroutine facet_normal_apply_surfvec(this, x, y, z, u, v, w, n, t, tstep)
146 class(facet_normal_t), intent(in) :: this
147 integer, intent(in) :: n
148 real(kind=rp), intent(inout), dimension(n) :: x
149 real(kind=rp), intent(inout), dimension(n) :: y
150 real(kind=rp), intent(inout), dimension(n) :: z
151 real(kind=rp), intent(inout), dimension(n) :: u
152 real(kind=rp), intent(inout), dimension(n) :: v
153 real(kind=rp), intent(inout), dimension(n) :: w
154 real(kind=rp), intent(in), optional :: t
155 integer, intent(in), optional :: tstep
156 integer :: i, m, k, idx(4), facet
157 real(kind=rp) :: normal(3), area
158
159 m = this%unique_mask(0)
160
161 do i = 1, m
162 k = this%unique_mask(i)
163 x(k) = u(k) * this%nx%x(i)
164 y(k) = v(k) * this%ny%x(i)
165 z(k) = w(k) * this%nz%x(i)
166 end do

222 subroutine facet_normal_finalize(this, only_facets)
223 class(facet_normal_t), target, intent(inout) :: this
224 logical, optional, intent(in) :: only_facets
225 logical :: only_facets_
226 type(htable_i4_t) :: unique_point_idx
227 integer :: htable_data, rcode, i, j, idx(4), facet
228 real(kind=rp) :: area, normal(3)
229
230 if (present(only_facets)) then
231 if (only_facets .eqv. .false.) then
232 call neko_error("For facet_normal_t, only_facets has to be true.")
233 end if
234 end if
235
236 call this%finalize_base(.true.)
237 ! This part is purely needed to ensure that contributions
238 ! for all faces a point is on is properly summed up.
239 ! If one simply uses the original mask, if a point is on a corner
240 ! where both faces are on the boundary
241 ! one will only get the contribution from one face, not both
242 ! We solve this by adding up the normals of both faces for these points
243 ! and storing this sum in this%nx, this%ny, this%nz.
244 ! As both contrbutions are added already,
245 ! we also ensure that we only visit each point once
246 ! and create a new mask with only unique points (this%unique_mask).
247 if (allocated(this%unique_mask)) then
248 deallocate(this%unique_mask)
249 end if
250 if (c_associated(this%unique_mask_d)) then
251 call device_free(this%unique_mask_d)
252 end if
253
254 call unique_point_idx%init(this%msk(0), htable_data)
255 j = 0
256 do i = 1, this%msk(0)
257 if (unique_point_idx%get(this%msk(i),htable_data) .ne. 0) then
258 j = j + 1
259 htable_data = j
260 call unique_point_idx%set(this%msk(i), j)
261 end if
262 end do
263 call this%nx%init(unique_point_idx%num_entries())
264 call this%ny%init(unique_point_idx%num_entries())
265 call this%nz%init(unique_point_idx%num_entries())
266 call this%work%init(unique_point_idx%num_entries())
267 allocate(this%unique_mask(0:unique_point_idx%num_entries()))
268
269 this%unique_mask(0) = unique_point_idx%num_entries()
270 do i = 1, this%unique_mask(0)
271 this%unique_mask(i) = 0
272 this%nx%x(i) = 0.0_rp
273 this%ny%x(i) = 0.0_rp
274 this%nz%x(i) = 0.0_rp
275 end do
276
277
278 do i = 1, this%msk(0)
279 rcode = unique_point_idx%get(this%msk(i), htable_data)
280 if (rcode .ne. 0) call neko_error("Facet normal: htable get failed.")
281 this%unique_mask(htable_data) = this%msk(i)
282 facet = this%facet(i)
283
284 idx = nonlinear_index(this%msk(i), this%Xh%lx, this%Xh%lx, this%Xh%lx)
285 normal = this%coef%get_normal(idx(1), idx(2), idx(3), idx(4), facet)
286 area = this%coef%get_area(idx(1), idx(2), idx(3), idx(4), facet)
287 normal = normal * area !Scale normal by area
288 this%nx%x(htable_data) = this%nx%x(htable_data) + normal(1)
289 this%ny%x(htable_data) = this%ny%x(htable_data) + normal(2)
290 this%nz%x(htable_data) = this%nz%x(htable_data) + normal(3)
291 end do
292
293 if (neko_bcknd_device .eq. 1) then
294 call device_map(this%unique_mask, this%unique_mask_d, &
295 size(this%unique_mask))
296 call device_memcpy(this%unique_mask, this%unique_mask_d, &
297 size(this%unique_mask), host_to_device, sync = .true.)
298 call device_memcpy(this%nx%x, this%nx%x_d, &
299 this%nx%n, host_to_device, sync = .true.)
300 call device_memcpy(this%ny%x, this%ny%x_d, &
301 this%ny%n, host_to_device, sync = .true.)
302 call device_memcpy(this%nz%x, this%nz%x_d, &
303 this%nz%n, host_to_device, sync = .true.)
304 end if
305
306 call unique_point_idx%free()
307
308 end subroutine facet_normal_finalize
